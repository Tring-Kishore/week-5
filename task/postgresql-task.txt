table accounts

CREATE TABLE Accounts(
account_id SERIAL PRIMARY KEY,
account_num VARCHAR(15) NOT NULL UNIQUE,
account_holder_name VARCHAR(50) NOT NULL,
balance DECIMAL(15,2) DEFAULT 0.00,
crated_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

table transacions

CREATE TABLE Transactions(
transaction_id SERIAL PRIMARY KEY,
account_num VARCHAR(15) NOT NULL,
transaction_type VARCHAR(20) NOT NULL ,
amount DECIMAL(15,2) NOT NULL,
transaction_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP ,
decription TEXT,
FOREIGN KEY (account_num) REFERENCES Accounts(account_num) ON DELETE CASCADE
);

DROP TABLE Transactions;

table transfer

CREATE TABLE transfer(
transfer_id SERIAL PRIMARY KEY ,
from_account_num VARCHAR(15) NOT NULL ,
to_account_num VARCHAR(15) NOT NULL ,
amount DECIMAL(15,2) NOT NULL,
transfer_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
status VARCHAR(20) DEFAULT 'pending',
FOREIGN KEY (from_account_num) REFERENCES Accounts(account_num) ON DELETE CASCADE,
FOREIGN KEY (to_account_num) REFERENCES Accounts(account_num) ON DELETE CASCADE
);

function to create account

CREATE OR REPLACE FUNCTION create_account(p_account VARCHAR(15),p_name VARCHAR(50),p_balance DECIMAL(15,2))
RETURNS VOID AS $$
BEGIN
IF LENGTH(p_account) <> 15 THEN
RAISE EXCEPTION 'Accout number should be 15 numbers';
END IF;
IF p_balance <= 0 THEN
RAISE EXCEPTION 'Amount should be greathan 0';
END IF;
INSERT INTO Accounts(account_num,account_holder_name,balance) VALUES (p_account,p_name,p_balance);
END;
$$ LANGUAGE plpgsql;

function for withdraw

CREATE OR REPLACE FUNCTION withdraw_amount(p_account VARCHAR(15),p_amount DECIMAL(15,2))
RETURNS VOID AS $$
DECLARE
v_balance DECIMAL(15,2);
BEGIN
SELECT balance INTO v_balance FROM Accounts WHERE account_num = p_account;
IF NOT FOUND THEN
RAISE EXCEPTION 'Account % not found',p_account;
END IF;
IF v_balance < p_amount THEN
RAISE EXCEPTION 'Insufficient Balance';
END IF;
UPDATE Accounts SET balance = balance - p_amount WHERE account_num = p_account;
INSERT INTO Transactions (account_num,transaction_type,amount) VALUES (p_account,'withdraw',p_amount);
END;
$$ LANGUAGE plpgsql;

function for deposit

CREATE OR REPLACE FUNCTION deposit_amount (p_account VARCHAR(15),p_amount DECIMAL(15,2))
RETURNS VOID AS $$
DECLARE
v_balance DECIMAL(15,2);
BEGIN
SELECT balance INTO v_balance FROM Accounts WHERE account_num = p_account;
IF NOT FOUND THEN
RAISE EXCEPTION 'Account % Not found',p_account;
END IF;
IF p_amount <= 0 THEN
RAISE EXCEPTION 'amount is negative';
END IF;
UPDATE Accounts SET balance = balance + p_amount WHERE account_num = p_account;
INSERT INTO Transactions (account_num,transaction_type,amount) VALUES (p_account,'deposit',p_amount);
END;
$$ LANGUAGE plpgsql;
